class Solution {
    public ListNode rotateRight(ListNode head, int k) {
        if(head == null || head.next == null || k == 0)
            return head;

        int n = 0;
        ListNode curr = head;

        while(curr != null) {
            n++;
            curr = curr.next;
        }

        k = k % n;

        if(k == 0)
            return head;

        curr = head;

        for(int i = 1; i < n - k; i++) {
            curr = curr.next;
        }

        ListNode newHead = curr.next;
        curr.next = null;

        ListNode tail = newHead;

        while(tail.next != null) {
            tail = tail.next;
        }

        tail.next = head;

        return newHead;
    }
}
