class Solution {
    public boolean isPalindrome(int x) {
        int temp = x;
        int rev = 0;
        while(temp > 0){
           int ld = temp % 10;
           rev = rev * 10 + ld;
           temp = temp / 10;
        }
        temp = x;
        if(temp == rev){
            return true;
        }
        else{
            return false;
        }
    }
}
