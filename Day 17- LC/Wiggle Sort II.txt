import java.util.*;

class Solution {
    public void wiggleSort(int[] nums) {
        int n = nums.length;

        Arrays.sort(nums);

        int[] temp = new int[n];

        int mid = (n - 1) / 2;
        int high = n - 1;

        for (int i = 0; i < n; i++) {
            if (i % 2 == 0) {
                temp[i] = nums[mid--];
            } else {
                temp[i] = nums[high--];
            }
        }

        for (int i = 0; i < n; i++) {
            nums[i] = temp[i];
        }
    }
}
